# ── Automatic dream cycle ────────────────────────────────────────────────────
# Dream cycles now fire on their own — no button. A background timer resets
# on every real chat send; if it ever runs out uninterrupted (nobody typing),
# dream.py is asked whether a pass is actually warranted, gated on both of
# the settings below. A manual override still exists (type "!dream" in the
# chat box) but is not shown anywhere in the UI.

# Seconds of no chat activity before a dream cycle is even considered. 0 or less turns automatic dreaming off
# (for example during a big import, or on a paid API where you want to choose when it runs); "!dream" still runs one.

# Minimum seconds since the last dream pass, regardless of idle time.
dream_min_interval_seconds = 7200

# Minimum new factual (non-derived) memories written since the last pass.
dream_min_new_memories = 20

# Seconds the dream rests before each of its model calls, to let a hot or shared GPU breathe. 0 = no rest.
